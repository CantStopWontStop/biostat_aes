lbw_data_groups <- lbw_data %>%
    mutate(
        age_group = case_when(
            age < 18 ~ "Underage",
            age < 35 & age >= 18 ~ "Young Adult",
            age >= 35 ~ "Adult",
            TRUE ~ NA_character_
        ),
        lwt_kg = round(lwt * 0.45359237),
        mother_weight_group = case_when(
            lwt_kg < 50 ~ "Underweight",
            lwt_kg < 70 & lwt_kg >= 50 ~ "Normal Weight",
            lwt_kg < 90 & lwt_kg >= 70 ~ "Overweight",
            lwt_kg >= 90 ~ "Obese"
        ),
        bwt_group = case_when(
            bwt < 2500 ~ "Underweight",
            bwt < 4000 & bwt >= 2500 ~ "Normal Weight",
            bwt >= 4000 ~ "Overweight"
        )
        
    ) %>% 
    mutate(age_group = factor(age_group, levels = c("Underage",
                                                    "Young Adult", 
                                                    "Adult")))
