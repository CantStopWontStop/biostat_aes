    status = case_when(
      outcome == 1 ~ 1L,                 # died
      TRUE ~ 0L                           # recovered or unknown -> treat as censored
    ),

    # Time origin: symptom onset (you can change the origin if your question differs)
    # Time-to-event in DAYS
    time = as.numeric(difftime(date_of_outcome, date_of_onset, units = "days"))
  ) %>%
  # Keep rows with a defined onset and non-negative time
  filter(!is.na(date_of_onset), !is.na(time), time >= 0) %>%
  mutate(
    gender = factor(gender, levels = c(0,1), labels = c("Female","Male")),
    hospital = factor(hospital),                 # treat as categorical site indicator
